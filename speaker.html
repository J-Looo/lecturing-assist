// 1. 下拉選單切換時，如果正在收音，強制重啟以套用新語言！ document.getElementById('sourceLangSelect').onchange = () => { const newLang = document.getElementById('sourceLangSelect').value; console.log("切換語音辨識語言為：", newLang); if (isListening && recognition) { recognition.stop(); // 先停止 setTimeout(() => { recognition.lang = newLang; // 指定新語言 try { recognition.start(); // 重新啟動 } catch(e) {} }, 300); } }; // 2. 麥克風按鈕點擊事件 document.getElementById('micBtn').onclick = () => { if (!recognition) return alert("瀏覽器不支援語音辨識！"); if (!isListening) { // 啟動「前」強制讀取當前下拉選單的值 const currentSelectedLang = document.getElementById('sourceLangSelect').value; recognition.lang = currentSelectedLang; try { recognition.start(); } catch(e) { console.error("啟動失敗", e); } } else { isListening = false; recognition.stop(); } };
